redis_client = Redis(host='localhost', port=6379, db=0)

def cache_fund_data(timeout=3600):
    def decorator(func):
        @wraps(func)
        def wrapper(fund_id):
            cache_key = f"fund:{fund_id}"
            cached = redis_client.get(cache_key)
            if cached:
                return json.loads(cached)
            
            result = func(fund_id)
            redis_client.setex(cache_key, timeout, json.dumps(result))
            return result
        return wrapper
    return decorator

async def get_funds(
    cursor: Optional[str] = None,
    limit: int = 50  # Max 50 per request
):
    query = db.query(MutualFund)
    
    if cursor:
        query = query.filter(MutualFund.id > cursor)
    
    funds = query.limit(limit + 1).all()
    
    has_more = len(funds) > limit
    if has_more:
        funds = funds[:-1]
    
    return {
        "data": funds,
        "next_cursor": funds[-1].id if has_more else None,
        "has_more": has_more
    }

# Add application monitoring
from prometheus_client import Counter, Histogram
import time

# Metrics
request_count = Counter('http_requests_total', 'Total HTTP requests', ['method', 'endpoint'])
request_duration = Histogram('http_request_duration_seconds', 'HTTP request duration')

@app.middleware("http")
async def monitor_requests(request, call_next):
    start_time = time.time()
    response = await call_next(request)
    
    duration = time.time() - start_time
    request_count.labels(method=request.method, endpoint=request.url.path).inc()
    request_duration.observe(duration)
    
    return response
